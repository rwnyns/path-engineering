// supabase/functions/chat/index.ts // يستقبل رسالة الطالب، يجيب الكورسات + آخر المحادثة، يسأل Gemini، يحفظ الرد. import { createClient } from "npm:@supabase/supabase-js@2"; const cors = { "Access-Control-Allow-Origin": "*", // بعد النشر غيّرها لدومين موقعك "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" } }); const MODEL = Deno.env.get("GEMINI_MODEL") ?? "gemini-flash-latest"; const MAX_LEN = 500; // أقصى طول لرسالة الطالب const HISTORY_LIMIT = 10; // كم رسالة سابقة نرسل للموديل const RATE_LIMIT = 10; // أقصى رسائل بالدقيقة للجلسة الواحدة const SYSTEM_PROMPT = `أنت مساعد منصة Path Engineering، منصة هندسية تخدم الطلاب والشركات والمرشدين. خدمات المنصة: (1) إدارة التدريب الميداني للشركات، (2) إرشاد مشاريع التخرج بالساعة مع مهندسين خبراء، (3) برنامج التمديد المهني لثلاثة أشهر للطلاب المتميزين. مهمتك: الإجابة عن أسئلة الطلاب بإيجاز ووضوح، واقتراح كورسات مناسبة لهم. القواعد: - اقترح كورسات من "قائمة الكورسات" فقط، ولا تخترع كورساً أو رابطاً أبداً. - إذا ما في كورس مناسب، قل ذلك بصراحة واقترح عليه التسجيل في المنصة ليتواصل معه الفريق. - إذا كان طلب الطالب غامضاً (التخصص أو المستوى)، اسأله سؤالاً واحداً قبل الاقتراح. - اذكر لكل كورس مقترح: العنوان، المستوى، المدة، والرابط إن وجد. لا تقترح أكثر من 3. - لا تذكر الأسعار أو وعوداً لا تعرفها؛ وجّه الطالب لاستمارة التسجيل لتفاصيل الباقات. - أجب بلغة الطالب (العربية افتراضياً) وبأسلوب ودّي ومختصر، بدون تنسيق markdown.`; Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); if (req.method !== "POST") return json({ error: "method not allowed" }, 405); try { const { message, session_id } = await req.json(); const text = String(message ?? "").trim().slice(0, MAX_LEN); const uuidRe = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i; if (!text || !uuidRe.test(String(session_id ?? ""))) return json({ error: "bad request" }, 400); const db = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!); // حماية بسيطة من الإساءة: حد أقصى للرسائل بالدقيقة const since = new Date(Date.now() - 60_000).toISOString(); const { count } = await db .from("chat_messages") .select("id", { count: "exact", head: true }) .eq("session_id", session_id).eq("role", "user").gte("created_at", since); if ((count ?? 0) >= RATE_LIMIT) return json({ reply: "رسائل كثيرة بوقت قصير، جرّب بعد دقيقة 🙏" }, 429); // الكورسات + آخر المحادثة بالتوازي const [coursesRes, historyRes] = await Promise.all([ db.from("courses").select("title, description, field, level, duration, url").eq("is_active", true).limit(100), db.from("chat_messages").select("role, content").eq("session_id", session_id) .order("created_at", { ascending: false }).limit(HISTORY_LIMIT), ]); const courseList = (coursesRes.data ?? []) .map((c) => `- ${c.title} | التخصص: ${c.field ?? "-"} | المستوى: ${c.level ?? "-"} | المدة: ${c.duration ?? "-"} | ${c.description ?? ""} | الرابط: ${c.url ?? "-"}`) .join("\n") || "(لا توجد كورسات حالياً)"; const history = (historyRes.data ?? []).reverse(); while (history.length && history[0].role !== "user") history.shift(); // Gemini يريد البداية user const contents = [ ...history.map((m) => ({ role: m.role === "bot" ? "model" : "user", parts: [{ text: m.content }] })), { role: "user", parts: [{ text }] }, ]; // حفظ رسالة الطالب (حتى لو فشل الموديل نعرف شو سأل) await db.from("chat_messages").insert({ session_id, role: "user", content: text }); const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent`, { method: "POST", headers: { "Content-Type": "application/json", "x-goog-api-key": Deno.env.get("GEMINI_API_KEY")! }, body: JSON.stringify({ systemInstruction: { parts: [{ text: `${SYSTEM_PROMPT}\n\nقائمة الكورسات:\n${courseList}` }] }, contents, generationConfig: { temperature: 0.5, maxOutputTokens: 700 }, }), }); if (!res.ok) throw new Error(`Gemini ${res.status}: ${await res.text()}`); const data = await res.json(); const reply = data?.candidates?.[0]?.content?.parts?.map((p: { text?: string }) => p.text ?? "").join("").trim() || "عذراً، ما قدرت أجاوب هلأ. جرّب صياغة ثانية."; await db.from("chat_messages").insert({ session_id, role: "bot", content: reply }); return json({ reply }); } catch (e) { console.error(e); return json({ error: "server error" }, 500); } });
